document.addEventListener('DOMContentLoaded', async () => { const detailContainer = document.getElementById('productDetail'); // ID kontainer detail produk // 1. Ambil ID produk dari URL (misal: detail.html?id=12) const urlParams = new URLSearchParams(window.location.search); const productId = urlParams.get('id'); if (!productId) { if (detailContainer) { detailContainer.innerHTML = '

Produk tidak ditemukan! ID tidak valid.

'; } return; } try { // 2. Fetch data dari API backend Vercel / Supabase const response = await fetch(`/api/products/detail?id=${productId}`); const result = await response.json(); if (!response.ok || !result.data) { throw new Error(result.message || 'Gagal memuat detail produk.'); } const product = result.data; // 3. Tampilkan data ke HTML if (detailContainer) { detailContainer.innerHTML = `

${product.nama || product.name}

Harga: Rp ${(product.harga || product.price || 0).toLocaleString('id-ID')}

Stok: ${product.stok || product.stock || 0} unit

${product.deskripsi || product.description || 'Tidak ada deskripsi produk.'}

`; } } catch (error) { console.error('Error loading product detail:', error); if (detailContainer) { detailContainer.innerHTML = `

Gagal memuat detail produk: ${error.message}

`; } } });
